CREATE TABLE PRODUCT_TYPE (
    PRODUCT_TYPE_ID INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    PRODUCT_TYPE_NAME VARCHAR(100) UNIQUE,
    PRODUCT_TYPE_COEFFICIENT DECIMAL(10,2)

-- Таблица типов материалов
CREATE TABLE MATERIAL_TYPE (
    MATERIAL_TYPE_ID INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    MATERIAL_TYPE_NAME VARCHAR(100) UNIQUE,
    PERCENTAGE_DEFECTIVE_MATERIAL DECIMAL(5,4)
);

-- Таблица продукции
CREATE TABLE PRODUCT (
    PRODUCT_ID INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    PRODUCT_ARTICLE VARCHAR(100),
    PRODUCT_NAME VARCHAR(150) UNIQUE,
    MINIMUM_COST_FOR_PARTNER DECIMAL(15,2),
    PRODUCT_TYPE_ID INT,
    MATERIAL_TYPE_ID INT,
    FOREIGN KEY (PRODUCT_TYPE_ID) REFERENCES PRODUCT_TYPE(PRODUCT_TYPE_ID),
    FOREIGN KEY (MATERIAL_TYPE_ID) REFERENCES MATERIAL_TYPE(MATERIAL_TYPE_ID)
);

-- Таблица цехов
CREATE TABLE WORKSHOP (
    WORKSHOP_ID INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    WORKSHOP_NAME VARCHAR(100) UNIQUE,
    WORKSHOP_TYPE VARCHAR(50),
    WORKERS_COUNT INT
);

-- Связующая таблица между продукцией и цехом
CREATE TABLE PRODUCT_WORKSHOP (
    PRODUCT_WORKSHOP_ID INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    PRODUCT_ID INT,
    WORKSHOP_ID INT,
    MANUFACTURING_TIME_HOURS DECIMAL(10,2),
    FOREIGN KEY (PRODUCT_ID) REFERENCES PRODUCT(PRODUCT_ID),

using FurnitureApp.Entities;
using System.Collections.Generic;
using System.Linq;

namespace FurnitureApp.Models;

public class ProductModel
{
    private readonly Product _product;
    private readonly List<ProductWorkshop> _workshops;
    
    public ProductModel(Product product)
    {
        _product = product;
        _workshops = product.ProductWorkshops.ToList();
    }
    
    public int ProductId => _product.ProductId;
    public string ArticleNumber => _product.ArticleNumber;
    public string ProductName => _product.ProductName;
    public string ProductType => _product.ProductType;
    public decimal MinPartnerPrice => _product.MinPartnerPrice;
    public string MainMaterial => _product.MainMaterial;
    
    public int TotalProductionTime => 
        (int)_workshops.Sum(w => w.ProductionTimeHours);
    
    public string WorkshopsList => 
        string.Join(", ", _workshops.Select(w => w.Workshop.WorkshopName));
}

using FurnitureApp.Entities;
using System.Linq;

namespace FurnitureApp.Services;

public static class ProductionCalculator
{
    public static int CalculateTotalProductionTime(Product product)
    {
        return (int)product.ProductWorkshops
            .Sum(pw => pw.ProductionTimeHours);
    }
    
    public static string GetWorkshopsList(Product product)
    {
        return string.Join(", ", 
            product.ProductWorkshops
                .Select(pw => pw.Workshop.WorkshopName)
                .OrderBy(name => name));
    }
}
Services/DatabaseService.cs
csharp
using FurnitureApp.Entities;
using Microsoft.EntityFrameworkCore;
using System.Collections.Generic;
using System.Linq;
using System.Threading.Tasks;

namespace FurnitureApp.Services;

public class DatabaseService
{
    private readonly DemoFdbContext _context;
    
    public DatabaseService(DemoFdbContext context)
    {
        _context = context;
    }
    
    public async Task<List<Product>> GetProductsAsync()
    {
        return await _context.Products
            .Include(p => p.ProductWorkshops)
            .ThenInclude(pw => pw.Workshop)
            .OrderBy(p => p.ProductName)
            .ToListAsync();
    }
    
    public async Task<Product?> GetProductByIdAsync(int id)
    {
        return await _context.Products
            .Include(p => p.ProductWorkshops)
            .ThenInclude(pw => pw.Workshop)
            .FirstOrDefaultAsync(p => p.ProductId == id);
    }
    
    public async Task<List<ProductType>> GetProductTypesAsync()
    {
        return await _context.ProductTypes.ToListAsync();
    }
    
    public async Task<List<MaterialType>> GetMaterialTypesAsync()
    {
        return await _context.MaterialTypes.ToListAsync();
    }
    
    public async Task SaveProductAsync(Product product)
    {
        if (product.ProductId == 0)
        {
            _context.Products.Add(product);
        }
        else
        {
            _context.Products.Update(product);
        }
        
        await _context.SaveChangesAsync();
    }
    
    public async Task DeleteProductAsync(Product product)
    {
        _context.Products.Remove(product);
        await _context.SaveChangesAsync();
    }
}

ViewModels/ProductsViewModel.cs
csharp
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using FurnitureApp.Entities;
using FurnitureApp.Models;
using FurnitureApp.Services;
using MsBox.Avalonia;
using MsBox.Avalonia.Enums;
using System.Collections.ObjectModel;
using System.Linq;
using System.Threading.Tasks;

namespace FurnitureApp.ViewModels;

public partial class ProductsViewModel : ObservableObject
{
    private readonly DatabaseService _dbService;
    private readonly MainWindowViewModel _mainViewModel;
    
    [ObservableProperty]
    private ObservableCollection<ProductModel> _products = new();
    
    [ObservableProperty]
    private ProductModel? _selectedProduct;
    
    public ProductsViewModel(DatabaseService dbService, MainWindowViewModel mainViewModel)
    {
        _dbService = dbService;
        _mainViewModel = mainViewModel;
        LoadProductsAsync();
    }
    
    private async Task LoadProductsAsync()
    {
        var products = await _dbService.GetProductsAsync();
        Products = new ObservableCollection<ProductModel>(
            products.Select(p => new ProductModel(p)));
    }
    
    [RelayCommand]
    private async Task AddProductAsync()
    {
        var productTypes = await _dbService.GetProductTypesAsync();
        var materialTypes = await _dbService.GetMaterialTypesAsync();
        
        var viewModel = new EditProductViewModel(
            _dbService,
            _mainViewModel,
            null,
            productTypes,
            materialTypes);
        
        _mainViewModel.ShowEditProductView(viewModel);
    }
    
    [RelayCommand]
    private async Task EditProductAsync()
    {
        if (SelectedProduct == null) return;
        
        var product = await _dbService.GetProductByIdAsync(SelectedProduct.ProductId);
        if (product == null) return;
        
        var productTypes = await _dbService.GetProductTypesAsync();
        var materialTypes = await _dbService.GetMaterialTypesAsync();
        
        var viewModel = new EditProductViewModel(
            _dbService,
            _mainViewModel,
            product,
            productTypes,
            materialTypes);
        
        _mainViewModel.ShowEditProductView(viewModel);
    }
    
    [RelayCommand]
    private async Task DeleteProductAsync()
    {
        if (SelectedProduct == null) return;
        
        var box = MessageBoxManager.GetMessageBoxStandard(
            "Подтверждение",
            $"Вы уверены, что хотите удалить продукт '{SelectedProduct.ProductName}'?",
            ButtonEnum.YesNo,
            Icon.Warning);
        
        var result = await box.ShowAsync();
        if (result != ButtonResult.Yes) return;
        
        var product = await _dbService.GetProductByIdAsync(SelectedProduct.ProductId);
        if (product != null)
        {
            await _dbService.DeleteProductAsync(product);
            await LoadProductsAsync();
        }
    }
}
ViewModels/EditProductViewModel.cs
csharp
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using FurnitureApp.Entities;
using FurnitureApp.Services;
using MsBox.Avalonia;
using MsBox.Avalonia.Enums;
using System.Collections.Generic;
using System.Linq;
using System.Threading.Tasks;

namespace FurnitureApp.ViewModels;

public partial class EditProductViewModel : ObservableObject
{
    private readonly DatabaseService _dbService;
    private readonly MainWindowViewModel _mainViewModel;
    private readonly Product? _originalProduct;
    
    [ObservableProperty]
    private string _articleNumber = string.Empty;
    
    [ObservableProperty]
    private string _productName = string.Empty;
    
    [ObservableProperty]
    private decimal _minPartnerPrice;
    
    [ObservableProperty]
    private string? _selectedProductType;
    
    [ObservableProperty]
    private string? _selectedMainMaterial;
    
    [ObservableProperty]
    private List<string> _productTypes = new();
    
    [ObservableProperty]
    private List<string> _materialTypes = new();
    
    [ObservableProperty]
    private string? _errorMessage;
    
    public EditProductViewModel(
        DatabaseService dbService,
        MainWindowViewModel mainViewModel,
        Product? product,
        List<ProductType> productTypes,
        List<MaterialType> materialTypes)
    {
        _dbService = dbService;
        _mainViewModel = mainViewModel;
        _originalProduct = product;
        
        ProductTypes = productTypes.Select(pt => pt.ProductTypeName).ToList();
        MaterialTypes = materialTypes.Select(mt => mt.MaterialTypeName).ToList();
        
        if (product != null)
        {
            ArticleNumber = product.ArticleNumber;
            ProductName = product.ProductName;
            MinPartnerPrice = product.MinPartnerPrice;
            SelectedProductType = product.ProductType;
            SelectedMainMaterial = product.MainMaterial;
        }
    }
    
    [RelayCommand]
    private async Task SaveAsync()
    {
        if (!Validate())
        {
            var box = MessageBoxManager.GetMessageBoxStandard(
                "Ошибка", 
                ErrorMessage, 
                ButtonEnum.Ok, 
                Icon.Error);
            
            await box.ShowAsync();
            return;
        }
        
        var product = _originalProduct ?? new Product();
        
        product.ArticleNumber = ArticleNumber;
        product.ProductName = ProductName;
        product.MinPartnerPrice = MinPartnerPrice;
        product.ProductType = SelectedProductType!;
        product.MainMaterial = SelectedMainMaterial!;
        
        await _dbService.SaveProductAsync(product);
        
        var successBox = MessageBoxManager.GetMessageBoxStandard(
            "Успех",
            "Продукт успешно сохранен",
            ButtonEnum.Ok,
            Icon.Success);
        
        await successBox.ShowAsync();
        
        _mainViewModel.ShowProductsView();
    }
    
    [RelayCommand]
    private void Cancel()
    {
        _mainViewModel.ShowProductsView();
    }
    
    private bool Validate()
    {
        if (string.IsNullOrWhiteSpace(ArticleNumber))
        {
            ErrorMessage = "Артикул не может быть пустым";
            return false;
        }
        
        if (string.IsNullOrWhiteSpace(ProductName))
        {
            ErrorMessage = "Название продукта не может быть пустым";
            return false;
        }
        
        if (MinPartnerPrice < 0)
        {
            ErrorMessage = "Цена не может быть отрицательной";
            return false;
        }
        
        if (string.IsNullOrEmpty(SelectedProductType))
        {
            ErrorMessage = "Необходимо выбрать тип продукта";
            return false;
        }
        
        if (string.IsNullOrEmpty(SelectedMainMaterial))
        {
            ErrorMessage = "Необходимо выбрать основной материал";
            return false;
        }
        
        ErrorMessage = null;
        return true;
    }
}

App.axaml.cs
csharp
using Avalonia;
using Avalonia.Controls.ApplicationLifetimes;
using Avalonia.Markup.Xaml;
using FurnitureApp.ViewModels;
using FurnitureApp.Views;

namespace FurnitureApp;

public partial class App : Application
{
    public override void Initialize()
    {
        AvaloniaXamlLoader.Load(this);
    }

    public override void OnFrameworkInitializationCompleted()
    {
        if (ApplicationLifetime is IClassicDesktopStyleApplicationLifetime desktop)
        {
            desktop.MainWindow = new MainWindow
            {
                DataContext = new MainWindowViewModel()
            };
        }

        base.OnFrameworkInitializationCompleted();
    }
}
