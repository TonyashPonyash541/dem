// DatabaseService.cs (Сервис работы с базой данных)
public static class DatabaseService
{
    private static string _connectionString = "DataSource=localhost;Port=3050;Database=/db/demo.fdb;Username=sysdba;Password=student";

    public static async Task<List<Product>> GetProducts()
    {
        var products = new List<Product>();

        using (var connection = new FbConnection(_connectionString))
        {
            await connection.OpenAsync();

            var command = new FbCommand(@"
                SELECT p.PRODUCT_ID, pt.PRODUCT_TYPE_NAME, p.PRODUCT_NAME, 
                       p.ARTICLE_NUMBER, p.MIN_PARTNER_PRICE, mt.MATERIAL_TYPE_NAME,
                       COALESCE(SUM(pw.PRODUCTION_TIME_HOURS), 0) AS TOTAL_PRODUCTION_TIME
                FROM PRODUCTS p
                JOIN PRODUCT_TYPE pt ON p.PRODUCT_TYPE_ID = pt.PRODUCT_TYPE_ID
                JOIN MATERIAL_TYPE mt ON p.MAIN_MATERIAL_ID = mt.MATERIAL_TYPE_ID
                LEFT JOIN PRODUCT_WORKSHOP pw ON p.PRODUCT_ID = pw.PRODUCT_ID
                GROUP BY p.PRODUCT_ID, pt.PRODUCT_TYPE_NAME, p.PRODUCT_NAME, 
                         p.ARTICLE_NUMBER, p.MIN_PARTNER_PRICE, mt.MATERIAL_TYPE_NAME
                ORDER BY p.PRODUCT_NAME", connection);

            using (var reader = await command.ExecuteReaderAsync())
            {
                while (await reader.ReadAsync())
                {
                    products.Add(new Product
                    {
                        ProductId = reader.GetInt32(0),
                        ProductType = reader.GetString(1),
                        ProductName = reader.GetString(2),
                        ArticleNumber = reader.GetString(3),
                        MinPartnerPrice = reader.GetDecimal(4),
                        MainMaterial = reader.GetString(5),
                        ProductionTime = (int)Math.Ceiling(reader.GetDecimal(6))
                    });
                }
            }
        }

        return products;
    }

    // Другие методы для работы с базой данных...
}

    {
        get => _product.ProductName;
        set => _product.ProductName = value;
    }

    public AddEditProductViewModel(Product product = null)
    {
        _product = product ?? new Product();
        _isEditMode = product != null;

        // Инициализация команд и загрузка данных
        InitializeCommands();
        LoadData();
    }

    private void InitializeCommands()
    {
        SaveCommand = ReactiveCommand.CreateFromTask(SaveProduct);
        CancelCommand = ReactiveCommand.Create(CloseWindow);
    }

    private async Task SaveProduct()
    {
        try
        {
            // Валидация данных
            if (string.IsNullOrWhiteSpace(ProductName))
            {
                await MessageBoxManager.GetMessageBoxStandardWindow(
                    "Ошибка", 
                    "Введите название продукта", 
                    ButtonEnum.Ok, 
                    Icon.Error).Show();
                return;
            }

            // Сохранение в базу данных
            if (_isEditMode)
            {
                await DatabaseService.UpdateProduct(_product);
            }
            else
            {
                await DatabaseService.AddProduct(_product);
            }

            CloseWindow();
        }
        catch (Exception ex)
        {
            await MessageBoxManager.GetMessageBoxStandardWindow(
                "Ошибка", 
                $"Не удалось сохранить продукт: {ex.Message}", 
                ButtonEnum.Ok, 
                Icon.Error).Show();
        }
    }

    private void CloseWindow()
    {
        (this as ICloseable)?.Close();
    }

    private async void LoadData()
    {
        // Загрузка типов продуктов и материалов для выпадающих списков
        try
        {
            ProductTypes = new ObservableCollection<string>(await DatabaseService.GetProductTypes());

                "Ошибка", 
                $"Не удалось загрузить данные: {ex.Message}", 
                ButtonEnum.Ok,

Метод расчета сырья
csharp
public class MaterialCalculator
{
    /// <summary>
    /// Рассчитывает количество сырья, необходимого для производства продукции
    /// </summary>
    /// <param name="productTypeId">Идентификатор типа продукции</param>
    /// <param name="materialTypeId">Идентификатор типа материала</param>
    /// <param name="productCount">Количество продукции</param>
    /// <param name="param1">Первый параметр продукции</param>
    /// <param name="param2">Второй параметр продукции</param>
    /// <returns>Количество сырья с учетом потерь или -1 при ошибке</returns>
    public static int CalculateRequiredMaterial(
        int productTypeId, 
        int materialTypeId, 
        int productCount, 
        double param1, 
        double param2)
    {
        // Проверка входных параметров
        if (productTypeId <= 0 || materialTypeId <= 0 || productCount <= 0 || param1 <= 0 || param2 <= 0)
        {
            return -1;
        }

        try
        {
            // Получаем данные из базы
            var (productTypeCoeff, materialLossPercentage) = GetCoefficients(productTypeId, materialTypeId);
            
            if (productTypeCoeff < 0 || materialLossPercentage < 0)
            {
                return -1; // Не найдены коэффициенты
            }

            // Расчет сырья на одну единицу продукции
            double materialPerUnit = param1 * param2 * productTypeCoeff;
            
            // Расчет общего сырья с учетом потерь
            double totalMaterial = materialPerUnit * productCount * (1 + materialLossPercentage / 100);
            
            // Округление в большую сторону
            return (int)Math.Ceiling(totalMaterial);
        }
        catch
        {
            return -1;
        }
    }

    private static (double productTypeCoeff, double materialLossPercentage) GetCoefficients(
        int productTypeId, 
        int materialTypeId)
    {
        using (var connection = new FbConnection(_connectionString))
        {
            connection.Open();

            var command = new FbCommand(@"
                SELECT pt.PRODUCT_TYPE_COEFFICIENT, mt.MATERIAL_LOSS_PERCENTAGE
                FROM PRODUCT_TYPE pt, MATERIAL_TYPE mt
                WHERE pt.PRODUCT_TYPE_ID = @productTypeId 
                AND mt.MATERIAL_TYPE_ID = @materialTypeId", connection);

            command.Parameters.AddWithValue("@productTypeId", productTypeId);
            command.Parameters.AddWithValue("@materialTypeId", materialTypeId);

            using (var reader = command.ExecuteReader())
            {
                if (reader.Read())
                {
                    return (reader.GetDouble(0), reader.GetDouble(1));
                }
            }
        }

        return (-1, -1);
    }
}

public class MainWindowViewModel : ReactiveObject
{
    private ObservableCollection<Product> _products = new();
    public ObservableCollection<Product> Products
    {
        get => _products;
        set => this.RaiseAndSetIfChanged(ref _products, value);
    }

    public ReactiveCommand<Unit, Unit> LoadProductsCommand { get; }

    public MainWindowViewModel()
    {
        LoadProductsCommand = ReactiveCommand.CreateFromTask(LoadProducts);
    }

    private async Task LoadProducts()
    {
        Products = new ObservableCollection<Product>(await DatabaseService.GetProducts());
    }
}

using Avalonia;

public partial class App : Application
{
    public override void Initialize()
    {
        AvaloniaXamlLoader.Load(this);
    }

    public override void OnFrameworkInitializationCompleted()
    {
        if (ApplicationLifetime is IClassicDesktopStyleApplicationLifetime desktop)
        {
            desktop.MainWindow = new MainWindow();
        }

        base.OnFrameworkInitializationCompleted();
    }
}
