MaterialTypeViewModel.cs
csharp
using System.Collections.ObjectModel;
using System.Linq;
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using DemoApp.Entities;
using DemoApp.Views;
using Microsoft.EntityFrameworkCore;

namespace DemoApp.ViewModels
{
    public partial class MaterialTypeViewModel : ViewModelBase
    {
        private readonly DbDemoFdbContext _context;
        private readonly MainWindowViewModel _mainViewModel;

        [ObservableProperty]
        private ObservableCollection<MaterialType> _materialTypes = new();

        [ObservableProperty]
        private MaterialType? _selectedMaterialType;

        public MaterialTypeViewModel(DbDemoFdbContext context, MainWindowViewModel mainViewModel)
        {
            _context = context;
            _mainViewModel = mainViewModel;
            LoadMaterialTypes();
        }

        public void LoadMaterialTypes()
        {
            MaterialTypes = new ObservableCollection<MaterialType>(_context.MaterialTypes.ToList());
        }

        [RelayCommand]
        private void AddMaterialType()
        {
            var editView = new EditMaterialTypeView
            {
                DataContext = new EditMaterialTypeViewModel(_context, this)
            };
            _mainViewModel.CurrentView = editView;
        }

        [RelayCommand]
        private void EditMaterialType()
        {
            if (SelectedMaterialType == null) return;
            
            var editView = new EditMaterialTypeView
            {
                DataContext = new EditMaterialTypeViewModel(_context, this, SelectedMaterialType)
            };
            _mainViewModel.CurrentView = editView;
        }
    }
}

using Microsoft.EntityFrameworkCore;
using DemoApp.Entities;

namespace DemoApp
{
    public class DbDemoFdbContext : DbContext
    {
        public DbSet<Material> Materials { get; set; }
        public DbSet<MaterialType> MaterialTypes { get; set; }
        public DbSet<Product> Products { get; set; }
        public DbSet<ProductType> ProductTypes { get; set; }
        public DbSet<Workshop> Workshops { get; set; }

        protected override void OnConfiguring(DbContextOptionsBuilder optionsBuilder)
        {
            optionsBuilder.UseFirebird("User=sysdba;Password=masterkey;Database=/db/demo.fdb;DataSource=localhost;Port=3050;");
        }

        protected override void OnModelCreating(ModelBuilder modelBuilder)
        {
            // Настройка связей между таблицами
            modelBuilder.Entity<Material>()
                .HasOne(m => m.MaterialTypeNavigation)
                .WithMany()
                .HasForeignKey(m => m.MaterialType);
        }
    }
}

2. EditMaterialTypeViewModel.cs (ViewModels)
csharp
using System.Threading.Tasks;
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using DemoApp.Entities;
using DemoApp.Helpers;
using MsBox.Avalonia;
using MsBox.Avalonia.Enums;

namespace DemoApp.ViewModels
{
    public partial class EditMaterialTypeViewModel : ViewModelBase
    {
        private readonly DbDemoFdbContext _context;
        private readonly MaterialTypeViewModel _parentViewModel;
        private readonly MaterialType _materialType;
        
        [ObservableProperty]
        private string _materialTypeName;
        
        [ObservableProperty]
        private decimal _materialLossPercentage;
        
        [ObservableProperty]
        private string? _errorMessage;

        public EditMaterialTypeViewModel(DbDemoFdbContext context, MaterialTypeViewModel parentViewModel, MaterialType? materialType = null)
        {
            _context = context;
            _parentViewModel = parentViewModel;
            _materialType = materialType ?? new MaterialType();
            
            MaterialTypeName = _materialType.MaterialTypeName;
            MaterialLossPercentage = _materialType.MaterialLossPercentage;
        }

        [RelayCommand]
        private async Task SaveAsync()
        {
            if (!Validate())
            {
                var box = MessageBoxManager.GetMessageBoxStandard("Ошибка", ErrorMessage, ButtonEnum.Ok);
                await box.ShowAsync();
                return;
            }

            _materialType.MaterialTypeName = MaterialTypeName;
            _materialType.MaterialLossPercentage = MaterialLossPercentage;

            if (_materialType.MaterialTypeId == 0)
            {
                _context.MaterialTypes.Add(_materialType);
            }

            await _context.SaveChangesAsync();
            _parentViewModel.LoadMaterialTypes();
            _parentViewModel._mainViewModel.ShowMaterialTypes();
        }

        [RelayCommand]
        private void Cancel()
        {
            _parentViewModel._mainViewModel.ShowMaterialTypes();
        }

        private bool Validate()
        {
            if (string.IsNullOrWhiteSpace(MaterialTypeName))
            {
                ErrorMessage = "Название типа материала обязательно";
                return false;
            }

            if (MaterialLossPercentage < 0 || MaterialLossPercentage > 100)
            {
                ErrorMessage = "Процент потерь должен быть между 0 и 100";
                return false;
            }

            ErrorMessage = null;
            return true;
        }
    }
}

using System.Text.RegularExpressions;

namespace DemoApp.Helpers
{
    public static class ValidationHelper
    {
        public static bool IsValidDecimal(string value, out decimal result)
        {
            return decimal.TryParse(value, out result) && result >= 0;
        }

        public static bool IsValidInt(string value, out int result)
        {
            return int.TryParse(value, out result) && result > 0;
        }

        public static bool IsValidString(string value, int maxLength = 100)
        {
            return !string.IsNullOrWhiteSpace(value) && value.Length <= maxLength;
        }
    }
